//===== rAthena Script ======================================= 
//= Warper RagnaRogue — Teleportadora Conquistada (Color-Coded)
//===== Descrição: ============================================
//= - Lista EXCLUSIVAMENTE mapas que foram conquistados/libertados na seed ativa.
//= - Mapas bloqueados pela névoa de guerra NÃO aparecem na lista.
//= - Color-Coded por grau de periculosidade:
//=   ^00BBFF [Azul]: Cidades Seguras
//=   ^00FF00 [Verde]: Lv 1-20 (Iniciante)
//=   ^3399FF [Azul]: Lv 21-45 (Intermediário)
//=   ^FF3333 [Vermelho]: Lv 46-75 (Perigo / Agressivo)
//=   ^FFFF00 [Amarelo]: Lv 76-99+ (Covil do Chefe / MVP)
//============================================================

-	script	Warper	-1,{

	if ($@current_world_seed$ == "") {
		.@rc = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		$@current_world_seed$ = (.@rc > 0 && .@seed$[0] != "") ? .@seed$[0] : "radiante-evil-druid-1649";
	}

	mes "[Teleportadora do Reino]";
	mes "Saudações, bravo aventureiro!";
	mes "As rotas arcanas conectam apenas os territórios ^00FF00libertados da névoa de guerra^000000.";
	mes "Mapas hostis ainda sob domínio inimigo não podem ser alcançados por teleporte.";
	next;

	// Contar mapas disponíveis
	.@town_count = query_sql("SELECT DISTINCT `map_name` FROM `world_map_conquests` WHERE `seed` = '" + $@current_world_seed$ + "' AND `map_name` IN ('prontera','geffen','morocc','payon','alberta','izlude','aldebaran','comodo','yuno','amatsu','gonryun','umbala','lighthalzen','louyang','ayothaya','einbroch','einbech','hugel','rachel','veins','xmas','jawaii') ORDER BY `map_name` ASC", .@town_maps$);
	.@dungeon_count = query_sql("SELECT DISTINCT `map_name` FROM `world_map_conquests` WHERE `seed` = '" + $@current_world_seed$ + "' AND `map_name` NOT IN ('prontera','geffen','morocc','payon','alberta','izlude','aldebaran','comodo','yuno','amatsu','gonryun','umbala','lighthalzen','louyang','ayothaya','einbroch','einbech','hugel','rachel','veins','xmas','jawaii') ORDER BY `conquered_at` DESC", .@combat_maps$);

	.@last_opt$ = (lastwarp$ != "") ? " ~ Último Teleporte (^0055FF" + lastwarp$ + "^000000)" : " ~ Último Teleporte ^777777[Nenhum]^000000";
	.@combat_label$ = (.@dungeon_count > 0) ? " ~ Áreas de Combate Purificadas (^00FF00" + .@dungeon_count + " mapas^000000)" : " ~ Áreas de Combate ^777777[Nenhum mapa purificado a pé ainda]^000000";

	.@main_choice = select(.@last_opt$, " ~ Cidades Seguras (^00BBFF" + .@town_count + " cidades^000000)", .@combat_label$, " ~ Cancelar");

	switch (.@main_choice) {
	case 1:
		if (lastwarp$ == "") {
			mes "[Teleportadora do Reino]";
			mes "Você ainda não viajou para nenhum mapa nesta sessão.";
			close;
		}
		callsub L_WarpToMap, lastwarp$;
		end;

	case 2:
		// Menu de Cidades Seguras Conquistadas
		if (.@town_count <= 0) {
			mes "[Teleportadora do Reino]";
			mes "Nenhuma cidade registrada no momento.";
			close;
		}
		.@menu$ = "";
		for (.@i = 0; .@i < .@town_count; .@i++) {
			.@m$ = .@town_maps$[.@i];
			.@menu$ = .@menu$ + "^00BBFF" + .@m$ + "^000000:";
		}
		.@menu$ = .@menu$ + "Cancelar";
		.@sel = select(.@menu$);
		if (.@sel > .@town_count) close;
		callsub L_WarpToMap, .@town_maps$[.@sel - 1];
		end;

	case 3:
		// Menu de Campos e Dungeons Conquistados (Color-Coded!)
		if (.@dungeon_count <= 0) {
			mes "[Teleportadora do Reino]";
			mes "Nenhum campo ou masmorra foi libertado ainda!";
			mes "Você deve marchar a pé e expulsar a vanguarda inimiga (15 abates ou 1 MVP) para purificar o mapa e registrá-lo nas rotas de teletransporte.";
			close;
		}
		.@menu$ = "";
		for (.@i = 0; .@i < .@dungeon_count; .@i++) {
			.@m$ = .@combat_maps$[.@i];
			.@color$ = callsub(L_GetMapColorTag, .@m$);
			.@menu$ = .@menu$ + .@color$ + .@m$ + "^000000:";
		}
		.@menu$ = .@menu$ + "Cancelar";
		.@sel = select(.@menu$);
		if (.@sel > .@dungeon_count) close;
		callsub L_WarpToMap, .@combat_maps$[.@sel - 1];
		end;

	case 4:
		close;
	}
	end;

L_WarpToMap:
	.@dest$ = getarg(0);
	lastwarp$ = .@dest$;

	// Buscar coordenadas seguras de ponto central ou portal de entrada
	.@x = 0;
	.@y = 0;
	.@sc = query_sql("SELECT `x`, `y` FROM `world_map_spawns` WHERE `map_name` = '" + escape_sql(.@dest$) + "' LIMIT 1", .@sx, .@sy);
	if (.@sc > 0) {
		.@x = .@sx[0];
		.@y = .@sy[0];
	}

	specialeffect2 568; // Luz Sagrada
	dispbottom "[Teleportadora] Teleportando com segurança para " + .@dest$ + "...", 0x00FF00;
	warp .@dest$, .@x, .@y;
	end;

L_GetMapColorTag:
	.@name$ = getarg(0);
	// Verde: Lv 1-20
	if (.@name$ == "prt_fild08" || .@name$ == "pay_fild08" || .@name$ == "gef_fild07" || .@name$ == "moc_fild12" || .@name$ == "iz_dun00" || .@name$ == "prt_sewb1" || .@name$ == "xmas_fild01" || .@name$ == "pay_fild01")
		return "^00FF00[Verde 1-20] ";
	// Azul: Lv 21-45
	if (.@name$ == "pay_dun01" || .@name$ == "gef_fild10" || .@name$ == "iz_dun02" || .@name$ == "moc_pryd01" || .@name$ == "ice_dun01" || .@name$ == "xmas_dun01" || .@name$ == "ein_fild08" || .@name$ == "mjo_dun01" || .@name$ == "prt_sewb2")
		return "^3399FF[Azul 21-45] ";
	// Vermelho: Lv 46-75
	if (.@name$ == "orcsdun02" || .@name$ == "pay_dun03" || .@name$ == "iz_dun04" || .@name$ == "moc_pryd04" || .@name$ == "in_sphinx4" || .@name$ == "ra_san01" || .@name$ == "ice_dun02" || .@name$ == "thor_v01" || .@name$ == "abyss_02" || .@name$ == "gl_prison" || .@name$ == "gl_church" || .@name$ == "nif_fild01" || .@name$ == "nameless_n")
		return "^FF3333[Vermelho 46-75] ";
	// Amarelo: Lv 76-99+ / Bosses
	return "^FFFF00[Amarelo Boss/High-End] ";

}
