//===== rAthena Script =======================================
//= RagnaRogue v3 — Sistema dos 7 Fragmentos do Cometa & Névoa de Guerra
//===== Descrição: ============================================
//= - 7 Fragmentos caídos do Cometa Negro geram a névoa maligna pelo reino.
//= - Cada fragmento foi devorado por um chefe cósmico (HP/ATK calibrados para 99).
//= - Para destruir o Fragmento, o golpe final DEVE ser com uma Relíquia Sagrada (Arma Tier 4).
//= - Mapas não purificados removem buffs positivos e aplicam miasma progressivo (1 debuff/minuto).
//= - Radar cósmico (Taxi Distance em saltos de portal no grafo) informa a proximidade do fragmento vivo.
//= - Comandos: @radar e @cometa
//============================================================

-	script	CometSystem	-1,{

OnInit:
	bindatcmd("radar", "CometSystem::OnRadarCommand", 0, 99);
	bindatcmd("cometa", "CometSystem::OnCometStatus", 0, 99);

	if ($@current_world_seed$ == "") {
		.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		if (.@row_count > 0 && .@seed$[0] != "") {
			$@current_world_seed$ = .@seed$[0];
		} else {
			$@current_world_seed$ = "v3-world";
		}
	}

	// Carregar e invocar todos os chefes de fragmentos ainda não derrotados
	.@frag_count = query_sql("SELECT `id`, `map_name`, `mob_id`, `mob_name` FROM `world_comet_fragments` WHERE `seed` = '" + $@current_world_seed$ + "' AND `defeated` = 0", .@ids, .@maps$, .@mobs, .@names$);
	
	for (.@i = 0; .@i < .@frag_count; .@i++) {
		killmonster .@maps$[.@i], "CometSystem::OnCometBossKill";
		monster .@maps$[.@i], 0, 0, .@names$[.@i], .@mobs[.@i], 1, "CometSystem::OnCometBossKill", 2;
	}
	end;

OnPCLoadMapEvent:
	.@map$ = strcharinfo(3);

	// Ignorar interiores, instâncias protegidas ou castelos
	if (compare(.@map$, "_in") || compare(.@map$, "cas") || compare(.@map$, "sec_in") || compare(.@map$, "job_") || compare(.@map$, "gld_") || compare(.@map$, "pvp_")) end;

	if ($@current_world_seed$ == "") {
		.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		if (.@row_count > 0 && .@seed$[0] != "") {
			$@current_world_seed$ = .@seed$[0];
		} else {
			$@current_world_seed$ = "v3-world";
		}
	}

	// -------------------------------------------------------------
	// 1. Radar de Proximidade (Taxi Distance no Grafo de Mapas)
	// -------------------------------------------------------------
	.@row_radar = query_sql("SELECT r.`distance_hops`, f.`id`, f.`mob_name`, f.`map_name`, f.`title`, f.`star_name` FROM `world_fragment_distances` r JOIN `world_comet_fragments` f ON r.`fragment_id` = f.`id` AND r.`seed` = f.`seed` WHERE r.`seed` = '" + $@current_world_seed$ + "' AND r.`from_map` = '" + .@map$ + "' AND f.`defeated` = 0 ORDER BY r.`distance_hops` ASC LIMIT 1", .@hops, .@frag_id, .@boss_name$, .@boss_map$, .@frag_title$, .@star_name$);

	if (.@row_radar > 0) {
		.@d = .@hops[0];
		if (.@d == 0) {
			mapannounce .@map$, "⚠️ [ALERTA CÓSMICO] VOCÊ ADENTROU O COVIL DO [" + .@frag_title$[0] + ": " + .@boss_name$[0] + "]!", bc_map, 0xFF0000;
			message strcharinfo(0), "=========================================================";
			message strcharinfo(0), "⚠️ [ALERTA CÓSMICO] VOCÊ ESTÁ NO MAPA DO " + .@frag_title$[0] + "!";
			message strcharinfo(0), "⚠️ O Chefe Cósmico [" + .@boss_name$[0] + "] domina esta região!";
			message strcharinfo(0), "⚠️ Apenas o golpe de uma Relíquia Sagrada (Arma Tier 4) pode ferir o Fragmento!";
			message strcharinfo(0), "=========================================================";
			specialeffect2 458;
			specialeffect2 568;

			// Iniciar debuff temático contínuo do Fragmento
			@active_frag_id = .@frag_id[0];
			deltimer "CometSystem::OnThematicDebuffTick";
			addtimer 3000, "CometSystem::OnThematicDebuffTick";
		} else if (.@d == 1) {
			dispbottom "⚡ [RADAR ESTELAR] O solo vibra intensamente com poeira cósmica...", 0xFFAA00;
			dispbottom "Há um Fragmento do Cometa [" + .@frag_title$[0] + ": " + .@boss_name$[0] + "] a apenas 1 portal de distância!", 0xFFFF55;
			specialeffect2 568;
			deltimer "CometSystem::OnThematicDebuffTick";
		} else if (.@d >= 2 && .@d <= 3) {
			dispbottom "🌌 [RADAR ESTELAR] Você sente a presença do [" + .@frag_title$[0] + "] vindo de " + .@d + " mapas adiante.", 0x00FFFF;
			deltimer "CometSystem::OnThematicDebuffTick";
		} else if (.@d >= 4 && .@d <= 5) {
			dispbottom "🌠 [RADAR ESTELAR] Um eco tênue de poeira cósmica ressoa no horizonte distante (" + .@d + " mapas até o " + .@frag_title$[0] + ")...", 0x88AAFF;
			deltimer "CometSystem::OnThematicDebuffTick";
		} else {
			deltimer "CometSystem::OnThematicDebuffTick";
		}
	}

	// -------------------------------------------------------------
	// 2. Sistema de Névoa & Miasma Tóxico
	// -------------------------------------------------------------
	.@conquered = 0;
	query_sql("SELECT count(*) FROM `world_map_conquests` WHERE `map_name` = '" + .@map$ + "' AND `seed` = '" + $@current_world_seed$ + "'", .@conquered);

	if (.@conquered > 0) {
		// Mapa já libertado: interrompe miasma e limpa efeitos
		killmonster .@map$, "CometSystem::OnMiasmaMobKill";
		deltimer "CometSystem::OnMiasmaTick";
		deltimer "CometSystem::OnThematicDebuffTick";
		@miasma_stage = 0;
		sc_end SC_SLOWDOWN;
		sc_end SC_BLIND;
		sc_end SC_HALLUCINATION;
		sc_end SC_CURSE;
		sc_end SC_BLEEDING;
		sc_end SC_DECREASEAGI;
		sc_start SC_EXPBOOST, 86400000, 50;
		sc_start SC_ITEMBOOST, 86400000, 50;
		dispbottom "a nevoa se dissipa e voce respira aliviado: este mapa está livre da maldição do cometa", 0x00FF00;
		end;
	}

	// Se for mapa não conquistado: remover bônus de purificação
	sc_end SC_EXPBOOST;
	sc_end SC_ITEMBOOST;

	// Se o mapa não tiver monstros cadastrados ou ativos, não pune com miasma
	.@mob_spawns = 0;
	query_sql("SELECT ifnull(max(`total_spawns`), 0) FROM `world_map_has_mobs` WHERE `map_name` = '" + .@map$ + "'", .@mob_spawns);
	if (.@mob_spawns == 0 && mobcount("this", "all") <= 0) {
		deltimer "CometSystem::OnMiasmaTick";
		deltimer "CometSystem::OnThematicDebuffTick";
		@miasma_stage = 0;
		end;
	}

	// -------------------------------------------------------------
	// 2.1 Invocar Monstros Agressivos da Névoa (Corrupção do Cometa)
	// -------------------------------------------------------------
	.@tier = 1;
	if (compare(.@map$, "fild") || compare(.@map$, "ship") || compare(.@map$, "sewb1") || compare(.@map$, "iz_dun00") || compare(.@map$, "iz_dun01")) {
		.@tier = 1;
	} else if (compare(.@map$, "dun01") || compare(.@map$, "dun02") || compare(.@map$, "pryd01") || compare(.@map$, "pryd02") || compare(.@map$, "anthell") || compare(.@map$, "sewb")) {
		.@tier = 2;
	} else if (compare(.@map$, "dun03") || compare(.@map$, "dun04") || compare(.@map$, "pryd04") || compare(.@map$, "sphinx") || compare(.@map$, "ra_san") || compare(.@map$, "ice_dun") || compare(.@map$, "thor_v01")) {
		.@tier = 3;
	} else {
		.@tier = 4;
	}

	if (mobcount(.@map$, "CometSystem::OnMiasmaMobKill") == 0) {
		if (.@tier == 1) {
			monster .@map$, 0, 0, "Familiar Sombrio", 1005, 10, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Zumbi da Névoa", 1015, 8, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Esqueleto Corrompido", 1076, 6, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Esporo Venenoso", 1077, 6, "CometSystem::OnMiasmaMobKill";
		} else if (.@tier == 2) {
			monster .@map$, 0, 0, "Múmia Espectral", 1041, 8, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Soldado Esqueleto", 1028, 8, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Carniçal Faminto", 1036, 6, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Drainliar Sanguinário", 1111, 8, "CometSystem::OnMiasmaMobKill";
		} else if (.@tier == 3) {
			monster .@map$, 0, 0, "Ísis Abissal", 1029, 8, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Múmia Anciã da Praga", 1297, 6, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Esqueleto Arqueiro", 1016, 8, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Minotauro Titânico", 1149, 6, "CometSystem::OnMiasmaMobKill";
		} else {
			monster .@map$, 0, 0, "Raydric Caído", 1163, 8, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Raydric Atirador", 1276, 6, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Cavaleiro do Abismo", 1219, 3, "CometSystem::OnMiasmaMobKill";
			monster .@map$, 0, 0, "Wraith Espectral", 1192, 6, "CometSystem::OnMiasmaMobKill";
		}
	}

	// -------------------------------------------------------------
	// 2.2 Alerta e Contagem do Miasma Letal (Morte em 3 Minutos)
	// -------------------------------------------------------------
	dispbottom "⚠️ [ALERTA DE MIASMA] A névoa tóxica do Cometa domina este território!", 0xFF3333;
	dispbottom "☠️ O ar é rarefeito e corrosivo. Purifique o mapa abatendo monstros ou você perecerá em 3 minutos!", 0xFFAA00;
	specialeffect2 458;

	@miasma_stage = 0;
	deltimer "CometSystem::OnMiasmaTick";
	addtimer 20000, "CometSystem::OnMiasmaTick";
	end;

OnThematicDebuffTick:
	.@map$ = strcharinfo(3);

	// Verificar se ainda há fragmento vivo neste mapa
	.@alive = 0;
	query_sql("SELECT count(*) FROM `world_comet_fragments` WHERE `seed` = '" + $@current_world_seed$ + "' AND `map_name` = '" + .@map$ + "' AND `defeated` = 0", .@alive);
	if (.@alive == 0) {
		deltimer "CometSystem::OnThematicDebuffTick";
		sc_end SC_SLOWDOWN;
		sc_end SC_BLIND;
		sc_end SC_HALLUCINATION;
		sc_end SC_CURSE;
		sc_end SC_BLEEDING;
		end;
	}

	// Aplicar debuff temático específico do chefe correspondente
	if (@active_frag_id == 1) {
		// 1: Baphomet (Flamejante) - Chamas Solares (drena HP + calor infernal)
		percentheal -2, 0;
		specialeffect2 431;
		dispbottom "[CHAMAS SOLARES] O calor cósmico do Fragmento Flamejante queima sua carne (-2% HP)!", 0xFF5533;
	} else if (@active_frag_id == 2) {
		// 2: Phreeoni (do Vazio) - Vácuo Gravitacional (drena SP + lentidão pesada)
		percentheal 0, -3;
		sc_start SC_SLOWDOWN, 7000, 1;
		specialeffect2 432;
		dispbottom "[VÁCUO GRAVITACIONAL] O vácuo espacial do Fragmento do Vazio drena sua energia (-3% SP) e dobra o peso dos seus passos!", 0xAA55FF;
	} else if (@active_frag_id == 3) {
		// 3: Eddga (Tempestuoso) - Tempestade Estelar (arcos elétricos e atraso de conjuração)
		sc_start SC_DECREASEAGI, 7000, 1;
		specialeffect2 429;
		dispbottom "[TEMPESTADE ESTELAR] Arcos voltaicos de eletricidade estática desestabilizam seus movimentos!", 0xFFFF33;
	} else if (@active_frag_id == 4) {
		// 4: Moonlight (Astral / Ilusão) - Névoa Hipnótica (alucinação ondulada e cegueira)
		sc_start SC_HALLUCINATION, 12000, 1;
		sc_start SC_BLIND, 8000, 1;
		specialeffect2 473;
		dispbottom "[NÉVOA HIPNÓTICA] A luz lunar do Fragmento distorce o espaço e cega sua visão!", 0x55FFFF;
	} else if (@active_frag_id == 5) {
		// 5: Garm (Glacial) - Zero Absoluto (nevasca polar cortante)
		sc_start SC_SLOWDOWN, 7000, 1;
		percentheal -1, 0;
		specialeffect2 433;
		dispbottom "[ZERO ABSOLUTO] O frio polar congelante do Fragmento Glacial gela seus músculos!", 0x88CCFF;
	} else if (@active_frag_id == 6) {
		// 6: Dark Lord (Meteoro / Corrupção) - Miasma Necrótico (maldição constante)
		sc_start SC_CURSE, 12000, 1;
		specialeffect2 443;
		dispbottom "[MIASMA NECRÓTICO] A corrupção cadavérica do Fragmento amaldiçoa sua força e sorte!", 0xFF0000;
	} else if (@active_frag_id == 7) {
		// 7: Lord of Death (Crepúsculo / Morte) - Marca da Condenação (sangramento e anulação de cura)
		sc_start SC_BLEEDING, 12000, 1;
		percentheal -2, -2;
		specialeffect2 440;
		dispbottom "[MARCA DA CONDENAÇÃO] A aura da morte consome seu sangue e enfraquece seu espírito!", 0xAA0000;
	}

	// Reagendar tick temático a cada 6 segundos enquanto estiver no mapa do chefe
	addtimer 6000, "CometSystem::OnThematicDebuffTick";
	end;

OnMiasmaTick:
	.@map$ = strcharinfo(3);

	// Se já foi conquistado ou saiu do mapa com névoa, para
	.@conquered = 0;
	query_sql("SELECT count(*) FROM `world_map_conquests` WHERE `map_name` = '" + .@map$ + "' AND `seed` = '" + $@current_world_seed$ + "'", .@conquered);
	if (.@conquered > 0) {
		@miasma_stage = 0;
		end;
	}

	@miasma_stage = @miasma_stage + 1;
	.@rem_sec = (9 - @miasma_stage) * 20;
	if (.@rem_sec < 0) .@rem_sec = 0;

	specialeffect2 458;

	if (@miasma_stage == 1) {
		dispbottom "☠️ [MIASMA - ESTÁGIO 1] A névoa se adensa! (-10% HP/SP, Lentidão) [" + .@rem_sec + "s restantes]", 0xFF8800;
		sc_start SC_SLOWDOWN, 25000, 1;
		percentheal -10, -10;
		addtimer 20000, "CometSystem::OnMiasmaTick";
	} else if (@miasma_stage == 2) {
		dispbottom "☠️ [MIASMA - ESTÁGIO 2] O ar infecta suas feridas! (-15% HP/SP, Sangramento) [" + .@rem_sec + "s restantes]", 0xFF6600;
		sc_start SC_BLEEDING, 25000, 1;
		percentheal -15, -15;
		addtimer 20000, "CometSystem::OnMiasmaTick";
	} else if (@miasma_stage == 3) {
		dispbottom "☠️ [MIASMA - ESTÁGIO 3] O peso da corrupção drena sua vitalidade! (-15% HP/SP, Maldição) [" + .@rem_sec + "s restantes]", 0xFF4400;
		sc_start SC_CURSE, 25000, 1;
		percentheal -15, -15;
		addtimer 20000, "CometSystem::OnMiasmaTick";
	} else if (@miasma_stage == 4) {
		dispbottom "☠️ [MIASMA - ESTÁGIO 4] As trevas fecham seus olhos! (-20% HP/SP, Cegueira) [" + .@rem_sec + "s restantes]", 0xFF2200;
		sc_start SC_BLIND, 25000, 1;
		percentheal -20, -20;
		addtimer 20000, "CometSystem::OnMiasmaTick";
	} else if (@miasma_stage == 5) {
		dispbottom "☠️ [MIASMA - ESTÁGIO 5] A névoa distorce a realidade! (-20% HP/SP, Alucinações) [" + .@rem_sec + "s restantes]", 0xFF0055;
		sc_start SC_HALLUCINATION, 25000, 1;
		percentheal -20, -20;
		addtimer 20000, "CometSystem::OnMiasmaTick";
	} else if (@miasma_stage == 6) {
		dispbottom "☠️ [MIASMA - ESTÁGIO 6] Seus membros perdem a força motriz! (-25% HP/SP, Desaceleração) [" + .@rem_sec + "s restantes]", 0xFF0033;
		sc_start SC_DECREASEAGI, 25000, 5;
		percentheal -25, -25;
		addtimer 20000, "CometSystem::OnMiasmaTick";
	} else if (@miasma_stage == 7) {
		dispbottom "☠️ [MIASMA CRÍTICO] ASFIXIA SEVERA! Seus pulmões ardem em brasa! (-30% HP/SP) [40s restantes!]", 0xFF0000;
		percentheal -30, -30;
		addtimer 20000, "CometSystem::OnMiasmaTick";
	} else if (@miasma_stage == 8) {
		dispbottom "☠️ [MIASMA TERMINAL] COLAPSO ORGÂNICO TOTAL! PURIFIQUE OU USE @SOS! (-35% HP/SP) [20s restantes!]", 0xFF0000;
		percentheal -35, -35;
		addtimer 20000, "CometSystem::OnMiasmaTick";
	} else {
		dispbottom "☠️ [MORTE POR ASFIXIA] O miasma corrosivo consumiu todo o seu ar vital!", 0xFF0000;
		specialeffect2 568;
		percentheal -100, -100;
		unitkill getcharid(3);
		@miasma_stage = 0;
	}
	end;

OnMiasmaMobKill:
	specialeffect2 568;
	end;

OnMapPurified:
	.@map$ = strcharinfo(3);
	killmonster .@map$, "CometSystem::OnMiasmaMobKill";
	deltimer "CometSystem::OnMiasmaTick";
	@miasma_stage = 0;
	sc_end SC_SLOWDOWN;
	sc_end SC_BLIND;
	sc_end SC_HALLUCINATION;
	sc_end SC_CURSE;
	sc_end SC_BLEEDING;
	sc_end SC_DECREASEAGI;
	sc_start SC_EXPBOOST, 86400000, 50;
	sc_start SC_ITEMBOOST, 86400000, 50;
	dispbottom "a nevoa se dissipa e voce respira aliviado: este mapa está livre da maldição do cometa", 0x00FF00;
	return;

OnCometBossKill:
	.@wlv = getequipweaponlv(EQI_HAND_R);
	.@map$ = strcharinfo(3);

	if ($@current_world_seed$ == "") {
		.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		if (.@row_count > 0 && .@seed$[0] != "") {
			$@current_world_seed$ = .@seed$[0];
		} else {
			$@current_world_seed$ = "v3-world";
		}
	}

	.@frag_found = query_sql("SELECT `id`, `mob_id`, `mob_name`, `title` FROM `world_comet_fragments` WHERE `seed` = '" + $@current_world_seed$ + "' AND `map_name` = '" + .@map$ + "' AND `defeated` = 0 LIMIT 1", .@fids, .@fmobs, .@fnames$, .@ftitles$);
	if (.@frag_found == 0) end;

	.@fid = .@fids[0];
	.@fmob = .@fmobs[0];
	.@fname$ = .@fnames$[0];
	.@ftitle$ = .@ftitles$[0];

	// VERIFICAÇÃO DA RELÍQUIA SAGRADA (ARMA TIER 4)
	if (.@wlv < 4) {
		// Não portava arma de nível 4! O chefe renasce da poeira estelar!
		monster .@map$, 0, 0, .@fname$, .@fmob, 1, "CometSystem::OnCometBossKill", 2;
		specialeffect2 458;
		dispbottom "=========================================================", 0xFF3333;
		dispbottom "[NÚCLEO INDESTRUTÍVEL] A carapaça cósmica absorveu o golpe comum!", 0xFF3333;
		dispbottom "O Fragmento Cósmico regenerou o corpo de " + .@fname$ + "!", 0xFF5555;
		dispbottom "Apenas uma Relíquia Sagrada (Arma de Nível 4) pode ferir o Fragmento!", 0xFFFF00;
		dispbottom "=========================================================", 0xFF3333;
		end;
	}

	// ABATE BEM-SUCEDIDO COM ARMA TIER 4!
	query_sql("UPDATE `world_comet_fragments` SET `defeated` = 1, `defeated_by` = '" + escape_sql(strcharinfo(0)) + "', `defeated_at` = NOW() WHERE `id` = " + .@fid + " AND `seed` = '" + $@current_world_seed$ + "'");

	// Purifica o mapa do boss automaticamente
	query_sql("INSERT IGNORE INTO `world_map_conquests` (`map_name`, `seed`, `conquered_by`, `conquered_at`) VALUES ('" + .@map$ + "', '" + $@current_world_seed$ + "', '" + escape_sql(strcharinfo(0)) + "', NOW())");

	// Limpar miasmas do herói
	callsub OnMapPurified;

	// Efeitos visuais e recompensa lendária
	specialeffect2 568;
	specialeffect2 569;
	specialeffect2 10;
	set Zeny, Zeny + 50000;
	getitem 603, 2; // 2x Velha Caixa Azul
	getitem 607, 3; // 3x Fruto de Yggdrasil
	getitem 969, 1; // 1x Ouro

	dispbottom "=========================================================", 0x00FF00;
	dispbottom "🌟 VOCÊ ESTILHAÇOU O FRAGMENTO DO COMETA COM SUA RELÍQUIA SAGRADA!", 0x00FF00;
	dispbottom "Recompensa da Coroa: +50.000 Zeny, Velhas Caixas Azuis e Frutos Sagrados!", 0xFFFF00;
	dispbottom "=========================================================", 0x00FF00;

	// Verificar quantos fragmentos restam vivos
	.@remaining = 0;
	query_sql("SELECT count(*) FROM `world_comet_fragments` WHERE `seed` = '" + $@current_world_seed$ + "' AND `defeated` = 0", .@remaining);

	if (.@remaining > 0) {
		announce "[ERA DA LUZ] O valoroso herói " + strcharinfo(0) + " empunhou sua Relíquia Sagrada (Arma Tier 4) e DESTRUIU o " + .@ftitle$ + " [" + .@fname$ + "] em " + .@map$ + "!", bc_all, 0xFFD700;
		announce "Ainda restam " + .@remaining + " Fragmentos da Névoa corrompendo Rune-Midgard! Sigam a pulsação do radar!", bc_all, 0xFFFF55;
		donpcevent "EventTelemetry::OnFragmentKill";
	} else {
		announce "👑👑👑 [SALVAÇÃO TOTAL DE RUNE-MIDGARD] TODOS OS 7 FRAGMENTOS DO COMETA FORAM DESTRUÍDOS!", bc_all, 0x00FF00;
		announce "O bravo herói " + strcharinfo(0) + " quebrou o último Fragmento! A Névoa Malígna foi dissipada de todas as terras para sempre!", bc_all, 0x00FFFF;
		donpcevent "EventTelemetry::OnFinalFragmentKill";
	}
	end;

OnRadarCommand:
	.@map$ = strcharinfo(3);

	if ($@current_world_seed$ == "") {
		.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		if (.@row_count > 0 && .@seed$[0] != "") {
			$@current_world_seed$ = .@seed$[0];
		} else {
			$@current_world_seed$ = "v3-world";
		}
	}

	.@c = query_sql("SELECT r.`distance_hops`, f.`mob_name`, f.`map_name`, f.`title` FROM `world_fragment_distances` r JOIN `world_comet_fragments` f ON r.`fragment_id` = f.`id` AND r.`seed` = f.`seed` WHERE r.`seed` = '" + $@current_world_seed$ + "' AND r.`from_map` = '" + .@map$ + "' AND f.`defeated` = 0 ORDER BY r.`distance_hops` ASC LIMIT 3", .@hops, .@bnames$, .@bmaps$, .@btitles$);

	if (.@c == 0) {
		dispbottom "[RADAR ESTELAR] O céu está calmo. Não há fragmentos ativos detectados ou todos foram destruídos!", 0x00FF00;
		end;
	}

	dispbottom "------ [RADAR DO COMETA NEGRO] ------", 0x00FFFF;
	for (.@i = 0; .@i < .@c; .@i++) {
		if (.@hops[.@i] == 0) {
			dispbottom "⚠️ #" + (.@i + 1) + " " + .@btitles$[.@i] + " (" + .@bnames$[.@i] + "): VOCÊ ESTÁ NO MESMO MAPA DO CHEFE!", 0xFF3333;
		} else {
			dispbottom "#" + (.@i + 1) + " " + .@btitles$[.@i] + " (" + .@bnames$[.@i] + "): " + .@hops[.@i] + " mapa(s) de distância via portais.", 0xFFFF88;
		}
	}
	dispbottom "-------------------------------------", 0x00FFFF;
	end;

OnCometStatus:
	if ($@current_world_seed$ == "") {
		.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		if (.@row_count > 0 && .@seed$[0] != "") {
			$@current_world_seed$ = .@seed$[0];
		} else {
			$@current_world_seed$ = "v3-world";
		}
	}

	.@total = query_sql("SELECT `id`, `mob_name`, `title`, `map_name`, `defeated`, ifnull(`defeated_by`, 'Nenhum') FROM `world_comet_fragments` WHERE `seed` = '" + $@current_world_seed$ + "' ORDER BY `id` ASC", .@ids, .@names$, .@titles$, .@maps$, .@defs, .@by$);

	dispbottom "====== [OS 7 FRAGMENTOS DO COMETA] ======", 0xFFAA00;
	for (.@i = 0; .@i < .@total; .@i++) {
		if (.@defs[.@i] == 1) {
			dispbottom "[DESTRUÍDO] " + .@titles$[.@i] + " (" + .@names$[.@i] + ") — Derrotado por: " + .@by$[.@i], 0x00FF00;
		} else {
			dispbottom "[ATIVO] " + .@titles$[.@i] + " (" + .@names$[.@i] + ") — Localização: " + .@maps$[.@i], 0xFF5555;
		}
	}
	dispbottom "=========================================", 0xFFAA00;
	end;
}
