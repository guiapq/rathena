//===== rAthena Script =======================================
//= RagnaRogue v3 — Sistema de Reconquista Territorial & Névoa de Guerra
//===== Descrição: ============================================
//= - Pune/bloqueia o uso do @warp e Teleportadora para mapas ainda não libertados a pé.
//= - Mapas com monstros exigem expulsar a vanguarda inimiga (15 abates ou 1 MVP) para purificar.
//= - Mapas sem monstros são conquistados automaticamente na primeira visita com mensagem especial.
//= - Emite anúncio global informando o nome do mapa e novos caminhos abertos.
//= - Registra o progresso do mundo procedural no MariaDB.
//============================================================

-	script	MapConquest	-1,{

OnInit:
	bindatcmd("warp", "MapConquest::OnWarpCommand", 0, 99);
	bindatcmd("rura", "MapConquest::OnWarpCommand", 0, 99);
	bindatcmd("mapmove", "MapConquest::OnWarpCommand", 0, 99);
	bindatcmd("go", "MapConquest::OnGoCommand", 0, 99);
	bindatcmd("sos", "MapConquest::OnSOSCommand", 0, 99);

	// Carregar seed ativa do mundo
	$@current_world_seed$ = "v3-world";
	.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
	if (.@row_count > 0 && .@seed$[0] != "") {
		$@current_world_seed$ = .@seed$[0];
	}

	// Apenas a Cidade Inicial ativa da seed e conquistada por padrao
	.@sc_rc = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'start_city' LIMIT 1", .@sc$);
	if (.@sc_rc > 0 && .@sc$[0] != "") {
		.@start_c$ = .@sc$[0];
	} else {
		.@start_c$ = "amatsu";
	}
	query_sql("INSERT IGNORE INTO `world_map_conquests` (`map_name`, `seed`, `conquered_by`, `conquered_at`) VALUES ('" + .@start_c$ + "', '" + $@current_world_seed$ + "', 'Sistema', NOW())");
	end;

OnPCLoadMapEvent:
	.@map$ = strcharinfo(3);
	getmapxy(.@cur_m$, .@cur_x, .@cur_y, 0);

	// Atualizar rastreamento do @sos (portal de entrada e mapa de origem)
	if (@sos_curr_map$ != .@cur_m$) {
		@sos_prev_map$ = @sos_curr_map$;
		@sos_prev_x = @sos_curr_x;
		@sos_prev_y = @sos_curr_y;
		@sos_curr_map$ = .@cur_m$;
		@sos_curr_x = .@cur_x;
		@sos_curr_y = .@cur_y;
	}

	// Se for cidade inicial ou interior/castelo/instancia protegida, ignorar
	if (compare(.@map$, "_in") || compare(.@map$, "cas") || compare(.@map$, "sec_in") || compare(.@map$, "job_") || compare(.@map$, "gld_") || compare(.@map$, "pvp_")) end;

	if ($@current_world_seed$ == "") {
		.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		if (.@row_count > 0 && .@seed$[0] != "") {
			$@current_world_seed$ = .@seed$[0];
		} else {
			$@current_world_seed$ = "v3-world";
		}
	}

	.@conquered = 0;
	query_sql("SELECT count(*) FROM `world_map_conquests` WHERE `map_name` = '" + .@map$ + "' AND `seed` = '" + $@current_world_seed$ + "'", .@conquered);
	if (.@conquered > 0) end;

	// Verificar se este mapa possui monstros cadastrados ou ativos
	.@mob_spawns = 0;
	query_sql("SELECT ifnull(max(`total_spawns`), 0) FROM `world_map_has_mobs` WHERE `map_name` = '" + .@map$ + "'", .@mob_spawns);
	.@active_mobs = mobcount("this", "all");

	if (.@mob_spawns == 0 && .@active_mobs <= 0) {
		// Mapas sem monstros são conquistados automaticamente com mensagem pacífica!
		query_sql("INSERT IGNORE INTO `world_map_conquests` (`map_name`, `seed`, `conquered_by`, `conquered_at`) VALUES ('" + .@map$ + "', '" + $@current_world_seed$ + "', '" + escape_sql(strcharinfo(0)) + "', NOW())");
		donpcevent "EventTelemetry::OnMapConquered";

		// Contar quantas novas rotas foram abertas a partir daqui
		.@new_maps_count = 0;
		query_sql("SELECT count(DISTINCT `to_map`) FROM `world_map_connections` WHERE `from_map` = '" + .@map$ + "' AND `seed` = '" + $@current_world_seed$ + "' AND `to_map` NOT IN (SELECT `map_name` FROM `world_map_conquests` WHERE `seed` = '" + $@current_world_seed$ + "')", .@new_maps_count);

		specialeffect2 568; // Coluna de Luz Sagrada
		dispbottom "[SANTUÁRIO PACÍFICO] A região de " + .@map$ + " não possui forças hostis e foi anexada aos territórios seguros!", 0x00FFFF;
		dispbottom "Fast Travel e rotas conectadas foram desbloqueados com sucesso!", 0x00FF00;

		// Mensagem diferenciada para mapas sem monstros
		if (.@new_maps_count > 0) {
			announce "Aventureiros, a pacífica região de " + .@map$ + " foi mapeada por " + strcharinfo(0) + ", " + .@new_maps_count + " novos mapas foram abertos!", bc_all, 0x00FFFF;
		} else {
			announce "Aventureiros, a pacífica região de " + .@map$ + " foi mapeada e integrada aos territórios livres por " + strcharinfo(0) + "!", bc_all, 0x00FFFF;
		}
		end;
	}

	// Se possui monstros: entrar em modo expedição e guerra de atrito
	@conquest_map_kills = 0;
	dispbottom "[EXPEDIÇÃO ROGUELIKE] Você adentrou " + .@map$ + "!", 0xFFAA00;
	dispbottom "Esta região está sob domínio inimigo. Elimine 15 monstros ou 1 MVP para expulsar a vanguarda e libertar as rotas!", 0xFFFF55;
	end;

OnNPCKillEvent:
	.@map$ = strcharinfo(3);

	// Cidades ou mapas especiais protegidos não contam
	if (compare(.@map$, "_in") || compare(.@map$, "cas") || compare(.@map$, "gld_") || compare(.@map$, "job_") || compare(.@map$, "sec_in")) end;

	if ($@current_world_seed$ == "") {
		.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		if (.@row_count > 0 && .@seed$[0] != "") {
			$@current_world_seed$ = .@seed$[0];
		} else {
			$@current_world_seed$ = "v3-world";
		}
	}

	.@conquered = 0;
	query_sql("SELECT count(*) FROM `world_map_conquests` WHERE `map_name` = '" + .@map$ + "' AND `seed` = '" + $@current_world_seed$ + "'", .@conquered);
	if (.@conquered > 0) end;

	// Abate de MVP purifica o mapa imediatamente
	.@is_mvp = 0;
	if (getmonsterinfo(killedrid, MOB_MVPEXP) > 0) {
		.@is_mvp = 1;
	}

	@conquest_map_kills = @conquest_map_kills + 1;

	if (.@is_mvp == 1 || @conquest_map_kills >= 15) {
		// Conquista de combate concluída com sucesso!
		query_sql("INSERT IGNORE INTO `world_map_conquests` (`map_name`, `seed`, `conquered_by`, `conquered_at`) VALUES ('" + .@map$ + "', '" + $@current_world_seed$ + "', '" + escape_sql(strcharinfo(0)) + "', NOW())");
		@conquest_map_kills = 0;
		donpcevent "CometSystem::OnMapPurified";
		donpcevent "EventTelemetry::OnMapConquered";

		// Quantos novos mapas ainda não conquistados foram abertos a partir deste mapa?
		.@new_maps_count = 0;
		query_sql("SELECT count(DISTINCT `to_map`) FROM `world_map_connections` WHERE `from_map` = '" + .@map$ + "' AND `seed` = '" + $@current_world_seed$ + "' AND `to_map` NOT IN (SELECT `map_name` FROM `world_map_conquests` WHERE `seed` = '" + $@current_world_seed$ + "')", .@new_maps_count);

		// Efeitos visuais e recompensa
		specialeffect2 568; // Coluna de Luz Sagrada
		specialeffect2 569;
		set Zeny, Zeny + 5000;
		dispbottom "Você purificou " + .@map$ + "! Recompensa da Coroa: +5.000 Zeny e Fast Travel desbloqueado!", 0x00FF00;

		// Anúncio global de combate
		if (.@new_maps_count > 0) {
			announce "Bravos guerreiros, " + .@map$ + " foi limpa por " + strcharinfo(0) + ", " + .@new_maps_count + " novos mapas foram abertos!", bc_all, 0x00FF00;
		} else {
			announce "Bravos guerreiros, " + .@map$ + " foi limpa por " + strcharinfo(0) + "!", bc_all, 0x00FF00;
		}
	} else {
		if (@conquest_map_kills % 3 == 0 || @conquest_map_kills >= 12) {
			dispbottom "[Vanguarda Inimiga] " + @conquest_map_kills + "/15 monstros eliminados em " + .@map$ + "...", 0xFFFF88;
		}
	}
	end;

OnWarpCommand:
	if (.@atcmd_numparameters < 1) {
		message strcharinfo(0), "Uso: @warp <nome_do_mapa> [x] [y]";
		end;
	}

	if ($@current_world_seed$ == "") {
		.@row_count = query_sql("SELECT `value` FROM `world_metadata` WHERE `key` = 'active_seed' LIMIT 1", .@seed$);
		if (.@row_count > 0 && .@seed$[0] != "") {
			$@current_world_seed$ = .@seed$[0];
		} else {
			$@current_world_seed$ = "v3-world";
		}
	}

	.@dest$ = .@atcmd_parameters$[0];
	.@x = (.@atcmd_numparameters > 1) ? atoi(.@atcmd_parameters$[1]) : 0;
	.@y = (.@atcmd_numparameters > 2) ? atoi(.@atcmd_parameters$[2]) : 0;

	// Ponto determinístico: se o jogador não especificou coordenadas (x=0 e y=0),
	// busca o centro da cidade ou o portal de entrada do mapa para evitar teleporte randômico
	if (.@x == 0 && .@y == 0) {
		.@spawn_found = query_sql("SELECT `x`, `y` FROM `world_map_spawns` WHERE `map_name` = '" + escape_sql(.@dest$) + "' LIMIT 1", .@sx, .@sy);
		if (.@spawn_found > 0) {
			.@x = .@sx[0];
			.@y = .@sy[0];
		}
	}

	.@conquered = 0;
	query_sql("SELECT count(*) FROM `world_map_conquests` WHERE `map_name` = '" + .@dest$ + "' AND `seed` = '" + $@current_world_seed$ + "'", .@conquered);

	if (.@conquered == 0) {
		if (getgmlevel() == 99) {
			message strcharinfo(0), "[NÉVOA DE GUERRA] Aviso: A região '" + .@dest$ + "' está bloqueada pela névoa de guerra! [GM 99 Bypass: Teleportando...]";
			warp .@dest$, .@x, .@y;
			end;
		}
		// Punição do @warp: teleporte bloqueado pela névoa de guerra!
		message strcharinfo(0), "=========================================================";
		message strcharinfo(0), "[NÉVOA DE GUERRA] A região '" + .@dest$ + "' ainda está sob domínio hostil!";
		message strcharinfo(0), "Você precisa marchar a pé e purificar o mapa antes de usar Fast Travel.";
		message strcharinfo(0), "=========================================================";
		specialeffect2 458; // Fumaça negra / debuff visual de bloqueio
		end;
	}

	// Mapa já conquistado: Fast Travel liberado com sucesso!
	message strcharinfo(0), "[Fast Travel] Teleportando para o mapa reconquistado: " + .@dest$;
	warp .@dest$, .@x, .@y;
	end;

OnGoCommand:
	// O comando @go leva para cidades seguras
	if (.@atcmd_numparameters < 1) {
		message strcharinfo(0), "Uso: @go <cidade/numero>";
		end;
	}
	return;

OnSOSCommand:
	getmapxy(.@my_m$, .@my_x, .@my_y, 0);

	// Se for executado em uma cidade, teleporta direto para o ponto central (@warp da cidade)
	.@is_town = query_sql("SELECT `x`, `y`, `note` FROM `world_map_spawns` WHERE `map_name` = '" + escape_sql(.@my_m$) + "' AND `type` = 'town' LIMIT 1", .@tx, .@ty, .@tnote$);
	if (.@is_town > 0) {
		message strcharinfo(0), "[SOS] Retornando ao ponto central de " + .@my_m$ + " (" + .@tnote$[0] + ")...";
		specialeffect2 568;
		warp .@my_m$, .@tx[0], .@ty[0];
		end;
	}

	// Se for executado em um interior de cidade (ex: prt_in, morocc_in), também retorna ao ponto central da cidade
	.@is_town_in = query_sql("SELECT s2.`map_name`, s2.`x`, s2.`y`, s2.`note` FROM `world_map_spawns` s1 JOIN `world_map_spawns` s2 ON s1.`note` = CONCAT('Portal de ', s2.`map_name`) WHERE s1.`map_name` = '" + escape_sql(.@my_m$) + "' AND s2.`type` = 'town' LIMIT 1", .@tm$, .@tx, .@ty, .@tnote$);
	if (.@is_town_in > 0) {
		message strcharinfo(0), "[SOS] Retornando ao ponto central de " + .@tm$[0] + " (" + .@tnote$[0] + ")...";
		specialeffect2 568;
		warp .@tm$[0], .@tx[0], .@ty[0];
		end;
	}

	// PROTOCOLO DE RESGATE DEFINITIVO (SOS):
	// Em qualquer campo ou masmorra hostil, evacua o jogador imediatamente para o SavePoint (cidade segura)
	message strcharinfo(0), "=========================================================";
	message strcharinfo(0), "🚨 [SOS RESGATE] Protocolo de Emergência acionado!";
	message strcharinfo(0), "Evacuando com segurança para o seu Ponto de Retorno (SavePoint)...";
	message strcharinfo(0), "=========================================================";

	deltimer "CometSystem::OnMiasmaTick";
	deltimer "CometSystem::OnThematicDebuffTick";
	@miasma_stage = 0;
	sc_end SC_SLOWDOWN;
	sc_end SC_BLIND;
	sc_end SC_HALLUCINATION;
	sc_end SC_CURSE;
	sc_end SC_BLEEDING;
	sc_end SC_DECREASEAGI;

	killmonster @sos_curr_map$, "CometSystem::OnMiasmaMobKill";
	specialeffect2 568;
	warp "SavePoint", 0, 0;
	end;
}
