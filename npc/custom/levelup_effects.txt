		dispbottom "[Marco Nível 80!] *Puf da Fúria Imparável!* Combo dos Titãs: Turbo Nitro & Fúria Selvagem!", 0x00FF88;
		end;
	}
	if (BaseLevel == 90) {
		specialeffect2 EF_TOPRANK;
		specialeffect2 EF_POK_LOVE;
		sc_start SC_ASSUMPTIO, 300000, 5;
		sc_start SC_TRUESIGHT, 240000, 10;
		sc_start SC_BLESSING, 300000, 10;
		sc_start SC_INCREASEAGI, 300000, 10;
		dispbottom "[Marco Nível 90 - Quase uma Lenda!] *Puf Campeão!* Combo dos 4 Elementos Sagrados ativado!", 0x00FF88;
		end;
	}
	if (BaseLevel >= 99) {
		specialeffect2 EF_MVP;
		specialeffect2 EF_LEVEL99_5;
		specialeffect2 EF_POKJUK;
		specialeffect2 EF_RAINBOW;
		sc_start SC_ASSUMPTIO, 600000, 5;
		sc_start SC_TRUESIGHT, 600000, 10;
		sc_start SC_BLESSING, 600000, 10;
		sc_start SC_INCREASEAGI, 600000, 10;
		sc_start SC_GLORIA, 600000, 5;
		sc_start SC_MAGNIFICAT, 600000, 5;
		sc_start SC_AURABLADE, 600000, 5;
		sc_start SC_KAIZEL, 600000, 7;
		announce "[LENDA ASCENDIDA] " + strcharinfo(0) + " alcançou o nível supremo 99 e recebeu as Bênçãos da Transcendência!", bc_all, 0xFFD700;
		dispbottom "[NÍVEL 99 MÁXIMO!] *PUF SUPREMO!* Você recebeu as 8 Bênçãos da Transcendência por 10 minutos!", 0x00FF88;
		end;
	}

	// 3. Níveis Regulares: Sorteia um buff garantidamente diferente do anterior
	.@idx = rand(.total_buffs);
	if (.@idx == @last_lvup_buff) {
		.@idx = (.@idx + 1 + rand(.total_buffs - 1)) % .total_buffs;
	}
	@last_lvup_buff = .@idx;

	// Aplica o buff positivo
	sc_start .buff_sc[.@idx], .buff_dur[.@idx], .buff_val[.@idx];

	// Notifica o jogador no chat
	dispbottom "[Level Up!] Nível " + BaseLevel + " alcançado! *Puf!* Você recebeu: " + .buff_names$[.@idx] + " (" + .buff_descs$[.@idx] + ")!", 0x00FF88;
	end;

OnPCJobLvUpEvent:
	// Brilho sutil dourado e mensagem ao subir o nível de classe
	specialeffect2 EF_BANJJAKII;
	dispbottom "[Classe Up!] Nível de Classe " + JobLevel + " alcançado!", 0xFFDD00;
	end;
